Eleven lines

The physics above, in short. Move your pointer over the source.

import { world } from 'anikit';

const w = world({ container: '#cards', gravity: [0, 1800] });

for (const card of cards) {
  // each card: six circles, rigidly linked
  const ids = card.cells.map(([x, y]) => w.add(x, y, r));
  pairs(ids).forEach(([a, b]) => w.link(a, b));
}

w.dragging();   // grab, swing and throw
